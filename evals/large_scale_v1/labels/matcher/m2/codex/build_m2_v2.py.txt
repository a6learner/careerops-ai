def build(batch, packet_path, output_path):
    with open(packet_path, encoding="utf-8") as handle:
        packet = json.load(handle)
    batch_codes = DECISION_CODES[batch]
    assert len(batch_codes) == len(packet)
    rendered = []
    for pair_index, (pair, pair_codes) in enumerate(zip(packet, batch_codes)):
        assert len(pair_codes) == len(pair["requirements"]), (batch, pair_index, len(pair_codes), len(pair["requirements"]))
        titles = "; ".join(exp["title"] for exp in pair["experiences"])
        labels = []
        for req_index, (req, code) in enumerate(zip(pair["requirements"], pair_codes)):
            key = (batch, pair_index, req_index)
            if code in {"Y", "P"}:
                assert key in POSITIVE, f"positive row lacks evidence: {key}"
                evs, rationale = POSITIVE[key]
                verdict = "matched" if code == "Y" else "partial"
                gap = None
            elif code == "I":
                assert key in INSUFFICIENT, f"insufficient row lacks reason: {key}"
                gap, rationale = INSUFFICIENT[key]
                verdict, evs = "insufficient_evidence", []
            elif code == "M":
                verdict, evs, gap = "missing", [], "required_technology_absent"
                rationale = f"The experiences ({titles}) do not explicitly prove this requirement: {req['text']}"
            else:
                raise AssertionError(f"unknown explicit code {code} at {key}")
            labels.append({
                "requirement_id": req["requirement_id"],
                "verdict": verdict,
                "evidence": evs,
                "gap_reason_code": gap,
                "rationale": rationale,
            })
        rendered.append({"pair_id": pair["pair_id"], "labeler": "codex", "labels": labels})
    with open(output_path, "w", encoding="utf-8") as handle:
        json.dump(rendered, handle, ensure_ascii=False, indent=2)
        handle.write("\n")


if __name__ == "__main__":
    build(int(sys.argv[1]), sys.argv[2], sys.argv[3])
